func TestReviewProjectionSwitchAndInvalidation(t *testing.T){
 db,_:=dbFor(t); svc:=practiceSvc(t,db);_,qv:=answer(t,db,svc)
 r,e:=revision.NewService(db,maintenance.NewService(db)).ApplyEvidenceDisposition(ctx,qv,"invalidate","review probe",nil);must(t,e)
 if r.Status!="succeeded"{t.Fatalf("rebuild: %+v",r)}
 var active string;must(t,db.QueryRow("SELECT active_version FROM projection_control").Scan(&active))
 var ord,norm,conf int
 must(t,db.QueryRow("SELECT coalesce(sum(ordinary_samples),0),coalesce(sum(normal_samples),0) FROM evidence_projection WHERE projection_version=?",active).Scan(&ord,&norm))
 must(t,db.QueryRow("SELECT count(*) FROM confusion_projection WHERE projection_version=?",active).Scan(&conf))
 t.Logf("after invalidation active=%s ordinary=%d normal=%d confusion_rows=%d",active,ord,norm,conf)
 answer(t,db,svc)
 var oldN,newN int
 must(t,db.QueryRow("SELECT coalesce(sum(ordinary_samples),0) FROM evidence_projection WHERE projection_version='v1'").Scan(&oldN))
 must(t,db.QueryRow("SELECT coalesce(sum(ordinary_samples),0) FROM evidence_projection WHERE projection_version=?",active).Scan(&newN))
 t.Logf("after new successful answer: v1 ordinary=%d active ordinary=%d",oldN,newN)
 if oldN!=2 || newN!=0 || norm!=1 || conf==0{t.Fatal("expected defects did not reproduce")}
}
